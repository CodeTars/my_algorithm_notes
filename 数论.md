## 幂

### 快速幂

```c++
typedef long long LL;  // 必要的时候可以用__int128
LL qpow(LL b, LL p, LL m)
{
    LL ans = 1;
    b %= m;
    for (; p; p >>= 1) {
        if (p & 1) {
            ans = (ans * b) % m;
        }
        b = (b * b) % m;
    }
    return ans;
}
```

### 矩阵快速幂

```c++
int N = 100;
const int MOD = 1e9 + 7;
struct matrix { LL m[105][105]; };

void SetOne(matrix& a)
{
    memset(a.m, 0, sizeof(a.m));
    for (int i = 0; i < N; a.m[i][i] = 1, i++);
}

matrix operator*(const matrix& a, const matrix& b)
{
    matrix c;
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            c.m[i][j] = 0;
            for (int k = 0; k < N; k++) {
                c.m[i][j] += (1LL * a.m[i][k] * b.m[k][j]) % MOD;
                c.m[i][j] %= MOD;
            }
        }
    }
    return c;
}

matrix mpow(matrix b, int n)
{
    matrix ans;
    SetOne(ans);
    for (; n; n >>= 1) {
        if (n & 1) {
            ans = ans * b;
        }
        b = b * b;
    }
    return ans;
}
```

## int128

__int128的输入和输出要自己实现，[A+B](https://www.luogu.com.cn/problem/P1601)

```c++
#include <cstdio>
#include <cctype>

__int128 read()
{
    char ch = '+';
    __int128 x = 0, f = 1;
    do {
        if (ch == '-') {
            f = -1;
        }
    } while (!isdigit(ch = getchar()));
    
    do {
        x = x * 10 + (ch - '0');
    } while (isdigit(ch = getchar()));
    return f * x;
}

void print(__int128 x)
{
    if (x) {
        if (x < 0) {
            putchar('-');
            x = -x;
        }
        print(x / 10);
        putchar('0' + x % 10);
    }
}

int main()
{
    __int128 a = read(), b = read(), c = a + b;
    if (c) {
        print(c);
    } else {
        putchar('0');
    }
    puts("");
    return 0;
}
```

## GCD  LCM

```c++
int gcd(int a, int b)
{
    return b ? gcd(b, a % b) : a;
}

int lcm(int a, int b)
{
    return a / gcd(a, b) * b;
}
```

## 素数

### 暴力判定素数

$O(\sqrt{n})$

```c++
bool isPrime(int n)
{
    if (n <= 1) {
        return false;
    } else {
        for (int i = 2; i <= sqrt(n); i++) {
            if (n % i == 0) {
                return false;
            }
        }
        return true;
    }
}
```

### 埃式筛法

$O(n\times \log(\log n))$

```c++
bool isPrime[int(1e8) + 1];

void Esieve(int n)
{
    vector<int> prime;
    memset(isPrime, true, sizeof isPrime);
    isPrime[0] = isPrime[1] = false;
    for (int i = 2; i <= sqrt(n); i++) {
        if (isPrime[i]) {
            for (int j = i * i; j <= n; j += i) {
                isPrime[j] = false;
            } 
        }
    }	
    for (int i = 2; i <= n; i++) {
        if (isPrime[i]) {
            prime.push_back(i);
        }
    }
    return prime;
}
```

### 线性筛法

$O(n)$

`c++`

```c++
bool isPrime[int(1e8) + 1];

vector<int> GetPrime(int n)
{
    memset(isPrime, true, sizeof isPrime);
    isPrime[0] = isPrime[1] = false;
 	
    vector<int> prime;   
    for (int i = 2; i <= n; i++) {
        if (isPrime[i]) {
            prime.push_back(i);
        }
        for (auto p : prime) {
            if (i * p > n) break;
            isPrime[i * p] = false;   
            if (i % p == 0) break;
        }
    }
	return prime;
}
```

`c`

```c++
const int N = 1e8;
const int NP = 1.3 * N / log(N);

bool isPrime[N + 19];
int prime[NP + 19], cnt = 0;

int GetPrime(int n)
{
    memset(isPrime, true, sizeof isPrime);
    isPrime[0] = isPrime[1] = false;
    
    for (int i = 2; i <= n; i++) {
        if (isPrime[i]) {
            prime[++cnt] = i;
        }
        for (int j = 1; j <= cnt && i * prime[j] <= n; j++) {
            isPrime[i * prime[j]] = false;   
            if (i % prime[j] == 0) break;
        }
    }
    
    return cnt;
}
```

### 质因数分解

$O(\sqrt{n})$

```c++
void factor(vector<int>& p, vector<int>& cnt, int n)
{
    for (int i = 2; i <= sqrt(n); i++) {
        if (n % i == 0) {
            p.push_back(i), cnt.push_back(0);
            while (n % i == 0) {
                n /= i, cnt.back()++;
            }
        }
    }

    if (n > 1) {
        p.push_back(n), cnt.push_back(1);
    }
}
```

## Gauss消元

### n元1次方程组

```c++
const double eps = 1e-6;

void Gauss(double a[N][N + 1], int n)
{
    for (int k = 1; k <= n - 1; k++) {
        int l = k;
        for (int i = k + 1; i <= n; i++) {
            if (fabs(a[i][k]) > fabs(a[l][k])) {
                l = i;
            }
        }
        if (l != k) {
            for (int j = 1; j <= n + 1; j++) {
                swap(a[k][j], a[l][j]);
            }
        }
        if (fabs(a[k][k]) < eps) {
            puts("No Solution");
            return;
        }
        for (int i = k + 1; i <= n; i++) {
            for (int j = k + 1; j <= n + 1; j++) {
                a[i][j] -= a[i][k] * a[k][j] / a[k][k];
            }
        }
    }
    
    if (fabs(a[n][n]) < eps) {
        puts("No Solution");
        return;
    }

    for (int i = n; i >= 1; i--) {
        double sum = 0;
        for (int j = i + 1; j <= n; j++) {
            sum += a[i][j] * a[j][n + 1];
        }
        a[i][n + 1] = (a[i][n + 1] - sum) / a[i][i];
    }

    for (int i = 1; i <= n; i++) {
        printf("%.2lf\n", a[i][n + 1]);
    }
}
```

### 异或方程组

```c++
void Gauss(int a[N][N + 1], int n)
{
    for (int k = 1; k <= n - 1; k++) {
        int l = k;
        for (; l <= n && a[l][k] == 0; l++);
        if (l > n) {
            puts("No Solution");
            return;
        }
        if (l != k) {
            for (int j = k; j <= n + 1; j++) {
                swap(a[l][j], a[k][j]);
            }
        }
        for (int i = k + 1; i <= n; i++) {
            if (a[i][k]) {
                for (int j = k + 1; j <= n + 1; j++) {
                    a[i][j] ^= a[k][j];
                }
            }
        }
    }

    if (a[n][n] == 0) {
        puts("No Solution");
        return;
    }
    
    for (int i = n; i >= 1; i--) {
        for (int j = i + 1; j <= n; j++) {
            a[i][n + 1] ^= (a[j][n + 1] & a[i][j]);
        }
    }
}
```

## 丢番图方程

### 扩展欧几里得

[青蛙的约会 ](http://poj.org/problem?id=1061) 丢番图方程

```c++
LL ExGcd(LL a, LL b, LL& x, LL& y)
{
    if (b) {
        LL d = ExGcd(b, a % b, y, x);
        y -= (a / b) * x;
        return d;
    } else {
        x = 1, y = 0;
        return a;
    }
}
```
